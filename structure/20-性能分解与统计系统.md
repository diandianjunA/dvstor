## 内容大纲

### 1. Breakdown系统架构 (`breakdown.hh`)
```cpp
namespace service::breakdown {

enum class Subcategory : u32 {
    // RDMA操作
    rdma_medoid_ptr, rdma_neighbor_fetch, rdma_vector_fetch,
    rdma_candidate_fetch, rdma_alloc, rdma_new_node_write,
    rdma_header_write, rdma_medoid_update, rdma_overflow_vec_fetch,
    rdma_neighbor_node_read, rdma_neighbor_lock, rdma_neighbor_unlock,
    rdma_neighbor_list_read, rdma_neighbor_list_write,
    rdma_pruned_neighbor_write,

    // GPU操作
    gpu_query_distance, gpu_insert_distance, gpu_insert_prune,
    gpu_insert_overflow_distance, gpu_insert_overflow_prune,

    // CPU操作
    cpu_query_select, cpu_query_filter, cpu_query_beam_update,
    cpu_query_beam_sort, cpu_query_result_ids, cpu_query_finalize,
    cpu_insert_init, cpu_insert_select, cpu_insert_filter,
    cpu_insert_beam_update, cpu_insert_preprune_sort,
    cpu_insert_candidate_collect, cpu_insert_candidate_sort,
    cpu_insert_prune_prepare, cpu_insert_pruned_neighbor_collect,
    cpu_insert_neighbor_prepare/collect, cpu_insert_overflow_prepare,
    cpu_insert_stage_candidates, cpu_insert_finalize, cpu_node_read,

    // 数据传输
    transfer_query_h2d, transfer_candidate_h2d, transfer_distance_d2h,
    transfer_insert_query_h2d, transfer_prune_h2d, transfer_prune_d2h,
    transfer_overflow_query_h2d, transfer_overflow_candidate_h2d,
    transfer_overflow_dist_d2h, transfer_overflow_prune_inputs_h2d,
    transfer_overflow_prune_d2h,
};

struct Sample {
    std::array<u64, NUM_SUBCATEGORIES> subcategory_times;
    u64 lock_attempts, lock_retries, cas_failures;
    u64 overflow_prune_max_candidates;
    void add_subcategory(Subcategory, u64 ns);
};

struct Report {
    // 聚合统计
    u64 query_count, insert_count;
    // 分位数
    u64 query_p50_ns, query_p95_ns, query_p99_ns;
    // ...
};
}
```

### 2. 统计系统 (`statistics.hh`)
```cpp
struct ThreadStatistics {
    u64 processed{0};           // 处理的任务总数
    u64 processed_queries{0};
    u64 processed_inserts{0};

    // 距离计算
    u64 distcomps{0};           // 总距离计算次数
    u64 query_distcomps{0};
    u64 build_distcomps{0};

    // RDMA统计
    u64 rdma_reads_in_bytes{0};
    u64 rdma_read_ops{0};
    u64 rdma_writes_in_bytes{0};
    u64 rdma_write_ops{0};
    // 按角色(insert/query) × 类型(neighbor/vector)细分

    // GPU内核调用
    u64 build_l2_kernels{0};
    u64 build_prune_kernels{0};

    // 数据传输
    u64 query_h2d_bytes{0}, query_d2h_bytes{0};
    u64 build_h2d_bytes{0}, build_d2h_bytes{0};

    // Overflow Prune统计
    u64 build_overflow_prunes{0};
    u64 build_overflow_prune_candidates{0};

    // 远程分配
    u64 remote_allocations{0};
    u64 allocation_size{0};
};
```

### 5. 性能瓶颈分析方法
基于分解数据的典型分析：
```
搜索延迟分解:
  rdma_medoid_ptr:      0.5%
  cpu_node_read:        0.2%
  rdma_neighbor_fetch: 15.3%   ← 主要RDMA开销
  cpu_query_filter:     2.1%
  rdma_vector_fetch:   25.7%   ← 最大开销（向量数据大）
  gpu_query_distance:   6.8%
  transfer_distance_d2h:5.2%
  cpu_query_beam_update:8.4%
  cpu_query_beam_sort:  4.1%
  cpu_query_result_ids:12.0%
  其他:                 19.7%

优化方向: 减少RDMA vector fetch开销
  → 量化存储（int8/uint8）
  → 更好的预取策略
  → 更大的beam合并RDMA请求
```
