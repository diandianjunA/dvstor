### 2. Float → 量化编码
```cpp
void encode_float_vector_to_storage(const float* src, u32 dim,
                                     VectorDType dst_dtype, byte_t* dst) {
    switch (dst_dtype) {
    case float32:
        memcpy(dst, src, dim * sizeof(float)); break;
    case uint8:
        for (i = 0; i < dim; i++)
            dst[i] = static_cast<uint8_t>(
                clamp((src[i] - min_val) / (max_val - min_val) * 255.0f, 0, 255));
        break;
    case int8:
        for (i = 0; i < dim; i++)
            dst[i] = static_cast<int8_t>(
                clamp((src[i] - min_val) / (max_val - min_val) * 255.0f - 128, -128, 127));
        break;
    }
}
```

### 3. 量化 → Float解码
```cpp
float vector_component_as_float(const byte_t* src, VectorDType src_dtype, size_t index) {
    switch (src_dtype) {
    case float32: return reinterpret_cast<const float*>(src)[index];
    case uint8:   return src[index] / 255.0f * (max - min) + min;
    case int8:    return (src[index] + 128) / 255.0f * (max - min) + min;
    }
}

vec<float> decode_storage_vector_to_float(const byte_t* src, VectorDType dtype, u32 dim);
```
